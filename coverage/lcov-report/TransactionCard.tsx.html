import React from 'react';
import { Box, HStack, VStack, Text, Icon, Pressable } from 'native-base';
import { Ionicons } from '@expo/vector-icons';
import { useTranslation } from 'react-i18next';
 
export type TransactionType = 'send' | 'receive' | 'qr_payment' | 'bill_payment' | 'topup';
export type TransactionStatus = 'completed' | 'pending' | 'failed';
 
export interface TransactionCardProps {
  type: TransactionType;
  status: TransactionStatus;
  amount: string;
  currency: string;
  recipient: string;
  date: string;
  description?: string;
  onPress?: () => void;
  testID?: string;
}
 
const getTransactionIcon = (type: TransactionType): keyof typeof Ionicons.glyphMap => {
  switch (type) {
    case 'send':
      return 'arrow-up-circle-outline';
    case 'receive':
      return 'arrow-down-circle-outline';
    case 'qr_payment':
      return 'qr-code-outline';
    case 'bill_payment':
      return 'receipt-outline';
    case 'topup':
      return 'add-circle-outline';
  }
};
 
const getStatusColor = (status: TransactionStatus): string => {
  switch (status) {
    case 'completed':
      return 'success.500';
    case 'pending':
      return 'warning.500';
    case 'failed':
      return 'error.500';
  }
};
 
const getAmountPrefix = (type: TransactionType): string => {
  switch (type) {
    case 'send':
    case 'qr_payment':
    case 'bill_payment':
      return '-';
    case 'receive':
    case 'topup':
      return '+';
  }
};
 
export const TransactionCard: React.FC<TransactionCardProps> = ({
  type,
  status,
  amount,
  currency,
  recipient,
  date,
  description,
  onPress,
  testID,
}) => {
  const { t } = useTranslation();
  const statusColor = getStatusColor(status);
  const amountPrefix = getAmountPrefix(type);
  const icon = getTransactionIcon(type);
 
  return (
    <Pressable
      onPress={onPress}
      testID={testID}
      _pressed={{ bg: 'gray.50' }}
    >
      <Box
        py={3}
        px={4}
        borderBottomWidth={1}
        borderBottomColor="gray.100"
      >
        <HStack space={4} alignItems="center">
          <Box
            p={2}
            rounded="full"
            bg={`${statusColor}:alpha.10`}
          >
            <Icon
              as={Ionicons}
              name={icon}
              size="md"
              color={statusColor}
              testID={`${testID}-icon`}
            />
          </Box>
          <VStack flex={1} space={1}>
            <HStack justifyContent="space-between" alignItems="center">
              <Text fontSize="md" fontWeight="semibold">
                {t(`transaction.type.${type}`)}
              </Text>
              <Text
                fontSize="md"
                fontWeight="bold"
                color={type === 'receive' || type === 'topup' ? 'success.500' : 'gray.800'}
                testID={`${testID}-amount`}
              >
                {amountPrefix}{currency} {amount}
              </Text>
            </HStack>
            <HStack justifyContent="space-between" alignItems="center">
              <VStack>
                <Text fontSize="sm" color="gray.600">
                  {recipient}
                </Text>
                {description && (
                  <Text fontSize="xs" color="gray.500" numberOfLines={1}>
                    {description}
                  </Text>
                )}
              </VStack>
              <Text fontSize="xs" color="gray.500">
                {date}
              </Text>
            </HStack>
            <Text
              fontSize="xs"
              color={statusColor}
              testID={`${testID}-status`}
            >
              {t(`transaction.status.${status}`)}
            </Text>
          </VStack>
        </HStack>
      </Box>
    </Pressable>
  );
};
